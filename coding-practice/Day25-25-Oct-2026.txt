Ansible - Setup Prometheus Node Exporter

- name: Setup Prometheus Node Exporter 
  hosts: all
  become: yes
  tasks:
    - name: Install Prometheus Node Exporter
      yum:
        name: prometheus-node-exporter
        state: present

    - name: Start Prometheus Node Exporter
      service:
        name: prometheus-node-exporter
        state: started
        enabled: yes

Terraform - Create Cloudwatch Alarms

resource "aws_cloudwatch_metric_alarm" "payment_api_cpu_alarm" {
    alarm_name              = "payment-api-cpu_utilization-high"
    comparison_operator     = "GreaterThanThreshold"
    evaluation_periods      = 1
    metric_name             = "CPUUtilization"
    namespace               = "AWS/EC2"
    period                  = 60
    statstics               = "Average"
    threshold               = 80

    dimension = {
        InstanceId          = aws_instance.payment_prod_ec2.id
    }
    alarm_actions           = [aws_sns_topic.payment_api_alerts.arn]
}

resource "aws_sns_topic" "payment_api_alerts" {
    name                    = "payment-api-alerts"
}

Kubernetes - Set UP Prometheus and Grafana

apiVersion: apps/v1
kind: Deployment
metadata:
  name: payment-api-prometheus
spec:
  replicas: 1
  selector:
    matchLabels:
      app: payment-api-prometheus
  template:
    metadata:
      labels:
        app: payment-api-prometheus
    spec:
      containers:
        - name: prometheus
          image: prom/prometheus:latest
          ports:
            - containerPort: 9090

JenkinsFile - Continuos Monitoring and Alerts

pipeline {
    agent any
    stages {
        stage('Monitor') {
            steps {
                script {
                    echo 'Publishing sample Cloudwatch metric'
                    sh 'aws cloudwatch put-metric-data --namespace "AWS/EC2" --metric-name "CPUUtilization" --value 90'
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
    }
}
