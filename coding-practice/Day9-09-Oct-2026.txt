Day 09 - Docker Integratin with Jenkins & GitLab
Focus: Docker Container Automation + AWS Auto Scaling + Kubernates HPA + Kubernates Deployment through CI/CD

Ansible - Setup Docker Container For Nginx
- name: Setup Docker Container for Nginx
  hosts: all
  become: yes
  tasks:
    - name: Pull Nginx Image
      docker_image:
        name: nginx
        source: pull
    - name: Run nginx Container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_port:
          - "8080:80"

Kubernates - Horizontal POD Autoscaler (HPA)
apiVersion: apps/v1
kind: Deployment
metadata: 
  name: payment-api
spec:
  replicas: 3
  selector:
    matchLabels:
      app: payment-api
  template:
    metadata:
      labels:
        app: payment-api
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80
            
---

apiVersion: autoscaling/v2
kind: HorozontalPodAutoscaller
metadata:
  name: payment-api-hpa
spec:
  scaleTargetRef:
    apiVersion: app/v1
    kind: Deployment
    name: payment-api
  minReplicas: 3
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50

JenkinsFile - Deploy to Kubernates
pipeline {
    agent any
    stages {
        stage('Deploy to Kubernates') {
            steps {
                script {
                    sh 'kubctl apply -f deployment.yaml'
                }
            }
        }
    }
}
