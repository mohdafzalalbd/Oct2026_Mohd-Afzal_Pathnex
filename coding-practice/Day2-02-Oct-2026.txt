Day 02 - Service & Tags

Ansible - Install & Enable Nginx

-name: Install & start Nginx on Production server
hosts: all
become: yes

tasks:
  -name: Install nginx
    yum:
      name: nginx
      state: present

  -name: Enable nginx
   service:
     name: nginx
     state: started
     enable: yes
     
Terraform - EC2 with Tags (r5.2xlarge)

  region = "us-east-1"
}

  ami         = "ami-0abchdjeb1272"

    name = "payment-prod-server"

Kubernetes - Deployment with 2 Replicas
apiversion: apps/v1
kind: deployment
metadata:
  name: payment-api
spec:
  replicas: 2
  selector:
    matchlabels:
      app: payment-api
  template:
    metadata:
      labels:
        app: payment-api
    spec:
      containers:
        -name: app
         image: nginx
         ports:
           -containerPort: 80
