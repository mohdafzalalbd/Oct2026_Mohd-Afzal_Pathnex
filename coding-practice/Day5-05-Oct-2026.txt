Jenkins file with Git integration and Testing

pipeline {
    agent any
    
    stages{
        stage("Clone Repository"){
            steps{
                git "https://github.com/payment-plateform/payment-api.git"
            }
        }
        stage("Build){
            steps{
                echo "Building project from git repository"
            }
        }
        stage("Test"){
            steps{
                echo "Running unit test"
            }
        }
        stage("Deploy"){
            steps{
                echo "Deploying Application"
            }
        }
    }
}

GitLab CI/CD - with Docker Build
stages:
 - build
 - push
 - deploy
 
 build:
   stage: build
   script:
     - docker build -t payment-api .
 push:
   stage: push
   script:
     - docker push payment-api
 deploy:
   stage: deploy
   script:
     - kubectl apply -f kubernates/deployment.yml

Ansible - Create user & Set Permission
- name: Create a new user
hosts: all
become: yes
tasks:
  -name: Create user
    user:
      name: payment-api
      state: present
      shell: /bin/bash
  -name: Set permission for the user
    file:
      path: /home/payment-api
      owner: payment-api
      group: payment-api
      mode: "0755"
