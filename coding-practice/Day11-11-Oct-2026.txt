Ansible - Install Helm on Server
- name: Install Helm
  hosts: all
  become: yes
  tasks:
    - name: Install Helm
      get_url:
        url: https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
        dest: /tmp/helm.tar.gz

    - name: Extract Helm
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes
        
    - name: Set Helm Binary Permissions
      file:
        path: /usr/local/bin/helm
        mode: '0755'

jenkinsFile - Multi-Stage Pipeline
pipeline {
  agent any
  environment {
    IMAGE_NAME = 'payment_api'
  }
  stages {
    stage('Checkout') {
      steps {
        git 'https://github.com/payment-platform/pyment-api.git'
      }
    }
    stage('Build Docker Image') {
      steps {
        script {
          docker.build("${env.IMAGE_NAME}")
        }
      }
    }
    stage('Push Docker Image') {
      steps {
        script {
          docker.withRegistry('https://docker.io', 'docker-credentials') {
            docker.image("${env.IMAGE_NAME}").push('latest')
          }
        }
      }
    }
    stage('Deploy to Kubernates') {
      steps {
        script {
          sh 'helm upgrade --install payment-api-nginx bitnami/nginx'
        }
      }
    }
  }
}

GitLab CI/CD - Build, Push & Deplow with Helm
stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "payment-api"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_NAME" .
  tags:
    - docker
push:
  stage: push
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" -u "$CI_REGISTRY_USER" --password-stdin
    - docker tag "$IMAGE_NAME" "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"
    - docker push "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA"
  tags:
    - docker
deploy:
  stage: deploy
  script:
    - helm repo add bitnami https://charts.bitnami.com/bitnami
    - helm repo update
    - helm upgrade --install payment-api-nginx bitnami/nginx --set image.registry="$CI_REGISTRY" --set image.repository="$CI_PROJECT_PATH/payment-api" --set image.tag="$CI_COMMIT_SHA"
  tags:
    - deploy
  rules:
    - if: '$CI_COMMIT_BRANCH == "master"'
      when: manual
