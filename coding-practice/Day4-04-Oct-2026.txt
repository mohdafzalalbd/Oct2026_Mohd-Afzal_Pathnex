Day 04 - Basic CICD concept with Jenkins & Gitlab Conditions:

Jenkinsfile - Basic CI Pipeline

pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building project'
            }
        }
        stage('Test') {
            steps {
                echo 'Runing tests'
            }
        }
    }
}

Gitlab CI/CD - Basic Pipeline

stages:
  - build
  - test
  - deploy
build:
  stage: build
  script:
    - echo "Building Project"
test:
  stage: test
  script:
    - echo "Running Test"
deploy:
  stage: deploy
  script:
    - echo "Deploying Application"

Ansible Install & configure Nginx

- name: Install & Configure Nginx
hosts: web_server
become: yes
  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present
    - name: Configure Nginx
      template:
        src: nginx.conf.j2
        dest: /etc/nginx/nginx.conf
    - name: Start Nginx Service
      service:
        name: nginx
        state: started
        enabled: yes

Terrafor EC2 with Security Group

resource "aws_security_group" "payment_api_sg" {
    name    ="payment-api-sg"
    description ="Allow SSH & HTTP"
    ingress {
        from_port = 22
        to_port   = 22
        protocol  = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    ingress {
        from_port   = 80
        to_port     = 80
        protocol    = "tcp"
        cidr_block  = ["0.0.0.0/0"]
    }

    egress {
        from_port   = 0
        to_port     = 0
        protocol    = "-1"
        cidr_block  = ["0.0.0.0/.0"]
    }
}
