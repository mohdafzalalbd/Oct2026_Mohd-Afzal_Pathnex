Ansible - install Docker

  hosts: all
  become: yes
  
  tasks:
    -name: Install Docker
      yum:
        name: docker
        state: present
    -name: Start Docker
      service:
        name: docker
        state: started
        enabled: yes

Kubernates - Pod with Volume Mount

apiVersion: v1
kind: Pod
Metadata:
  name: payment-api-pod

spec:
  containers:
    - name: web
      image: nginx
      volumeMounts:
        - mountPath: /usr/share/nginx/html
          name: html-volume
  volumes:
    - name: html-volume
      hostPath:
        path: /data
        type: Directory

Shell Script - Check running Services

#!/bin/bash
services=("nginx" "docker" "httpd")
for service in "${services[@]}"
do
    if systemctl is-active --quiet $service; then
      echo "$service is running"
    else
        echo "$service is not running"
    fi
done
